cmake_minimum_required(VERSION 3.22.1)
project(ReaderMobi C)
set(USE_ENCRYPTION OFF CACHE BOOL "" FORCE)
set(USE_LIBXML2 OFF CACHE BOOL "" FORCE)
set(USE_XMLWRITER ON CACHE BOOL "" FORCE)
set(USE_ZLIB ON CACHE BOOL "" FORCE)
set(LIBMOBI_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/libmobi")
add_subdirectory("${LIBMOBI_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/libmobi" EXCLUDE_FROM_ALL)
set_target_properties(mobi PROPERTIES VERSION "" SOVERSION "")
add_library(reader_mobi SHARED reader_mobi.c)
target_include_directories(reader_mobi PRIVATE "${LIBMOBI_DIR}/src")
target_link_libraries(reader_mobi PRIVATE mobi)
